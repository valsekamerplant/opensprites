async function extractAtlasStrip(atlasDataUrl: string, spec: PartSpec, spriteId: number): Promise<{ dataUrl: string; image: HTMLImageElement }> {
  const atlas = await loadImage(atlasDataUrl);
  const cols = Math.floor(atlas.width / FRAME_W);
  const rows = Math.floor(atlas.height / FRAME_H);
  const cell = Math.max(0, spriteId) * spec.frames;
  if (cols < spec.frames || rows < 1 || cell >= cols * rows) throw new Error(`Sprite ID ${spriteId} is outside this atlas.`);
  const canvas = document.createElement('canvas'); canvas.width = spec.frames * FRAME_W; canvas.height = FRAME_H;
  const ctx = canvas.getContext('2d')!; ctx.imageSmoothingEnabled = false;
  for (let frame = 0; frame < spec.frames; frame++) {
    const absolute = cell + frame;
    ctx.drawImage(atlas, (absolute % cols) * FRAME_W, Math.floor(absolute / cols) * FRAME_H, FRAME_W, FRAME_H, frame * FRAME_W, 0, FRAME_W, FRAME_H);
  }
  const dataUrl = canvasDataUrl(canvas);
  return { dataUrl, image: await loadImage(dataUrl) };
}

async function exportStandalonePack() {
  const project = readProject();
  const layers: Array<{ part: PartKey; atlas: string; frames: number; file: string }> = [];
  const files: Array<{ name: string; data: string | Uint8Array }> = [];
  for (const spec of profile().parts) {
    const strip = await partStrip(spec.key);
    if (!strip) {
      if (spec.optional || spec.specialOnly || project.item.equipmentType === 'projectile') continue;
      status.textContent = `Missing required ${spec.label} source.`;
      return;
    }
    const file = `sprites/${spec.atlas.replace(/\.png$/i, '')}-strip.png`;
    files.push({ name: file, data: dataUrlToBytes(canvasDataUrl(strip)) });
    layers.push({ part: spec.key, atlas: spec.atlas, frames: spec.frames, file });
  }
  const manifest = { format: 'openspell-sprite-pack', version: 1, item: project.item, layers, note: 'These strips are standalone authoring outputs. A full appearance.carbon is optional and only needed for a merged native atlas export.' };
  files.push({ name: 'itemdefs.carbon', data: JSON.stringify([itemDefinition(project)], null, 2) });
  files.push({ name: 'manifest.json', data: JSON.stringify(manifest, null, 2) });
  files.push({ name: 'README.txt', data: 'Standalone OpenSpell sprite pack\n\nThe sprites/ directory contains the authored 5- or 15-frame layer strips. Import appearance.carbon only when you want the studio to repack a complete native atlas.\n' });
  downloadBlob(buildZip(files), `${slugify(project.item.name)}-sprite-pack.zip`);
  status.textContent = 'Standalone sprite pack exported; no carbon bundle was required.';
}
